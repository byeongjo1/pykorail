```python
from datetime import datetime
from pykorail import Korail

with Korail.logged_in("me@example.com", "password") as korail:
    trains = korail.trains.search(
        "서울",
        "부산",
        depart_after=datetime(2026, 4, 1, 9, 0),
    )
    for train in trains:
        print(train)
```
